-- reconciliation: every ONS series -> one plant_id (SUCCESS #1)
SELECT match_method, count(*) AS ons_series, round(sum(g.mwh) / 1e6, 2) AS twh
      FROM ampera_catalog.ampera_silver.plant_match m
      LEFT JOIN (SELECT ons_key, sum(mwh) AS mwh FROM ampera_catalog.ampera_silver.generation GROUP BY 1) g USING (ons_key)
      GROUP BY 1 ORDER BY twh DESC
